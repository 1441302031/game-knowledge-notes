1. 架构概览

┌─────────────────────────────────────────────────┐
│                   Unity 客户端                    │
│  ┌─────────┐  ┌──────────┐  ┌───────────────┐  │
│  │ Network │  │  Match   │  │   Battle      │  │
│  │ Manager │  │  UI      │  │   (帧同步)     │  │
│  └────┬────┘  └──────────┘  └───────┬───────┘  │
│       │                             │           │
│       │  TCP + Protobuf             │ TCP直连   │
└───────┼─────────────────────────────┼───────────┘
        │                             │
   ┌────▼──────┐              ┌───────▼───────┐
   │  Gateway  │              │ Battle Server │
   │  :9000    │              │    :9100      │
   └────┬──────┘              └───────────────┘
        │ gRPC
   ┌────▼──────┐  ┌────────┐  ┌──────────┐
   │Game Server│  │  Room  │  │   Chat   │
   │  :50051   │  │ :50052 │  │  :50053  │
   └───────────┘  └────────┘  └──────────┘

关键概念（记住这 3 条就够了）：

概念 一句话解释
Gateway 统一入口，你永远只连它（除非进入战斗）
ClientEnvelope 所有消息都用这个信封包裹，里面写清楚"发给谁、什么类型"
战斗直连 匹配成功后，Gateway 告诉你 Battle 地址，你直接连过去

3. 第一步：Proto → C

3.1 协议文件一览

Ares 使用 Protobuf 协议，所有定义在 proto/ 目录。你只需要关心 3 个文件：

Proto 文件 用途 你发什么 你收什么
gateway.proto 消息信封 ClientEnvelope ServerEnvelope, RedirectNotify
common.proto 共享类型 Vec3, PlayerInfo 同
battle.proto 战斗协议 PlayerInput FrameData

3.2 生成 C# 代码

# 安装 protoc（如果没装）
# macOS: brew install protobuf
# Windows: 下载 https://github.com/protocolbuffers/protobuf/releases

# 生成 C# 代码
cd proto
protoc --csharp_out=../UnityProject/Assets/Scripts/Proto \
       --proto_path=. \
       common.proto gateway.proto battle.proto

生成后你会得到： - Common.cs — Vec3, SyncMode, PlayerInfo 等 - Gateway.cs — ClientEnvelope, ServerEnvelope, MsgCategory 等
- Battle.cs — PlayerInput, FrameData 等

// Common.cs
using Google.Protobuf;
using Google.Protobuf.Reflection;

public enum SyncMode {
    Lockstep = 0, StateSync = 1, Hybrid = 2
}

public class Vec3 {
    public float X, Y, Z;
}

public class BattlePlayerInfo {
    public ulong PlayerId;
    public string Nickname;
    public uint HeroId, TeamId, Slot;
}

// Gateway.cs
public enum MsgCategory {
    Auth = 0, Lobby = 1, Room = 2, Chat = 3, Battle = 4, System = 5
}

// ★ 你发的每条消息都包在这里
public class ClientEnvelope {
    public ulong PlayerId;
    public ulong Seq;
    public MsgCategory Category;
    public uint MsgId;
    public byte[] Payload; // ← 里面放具体消息的 protobuf bytes
    public ulong Timestamp;

    public byte[] Serialize() {
        using var ms = new MemoryStream();
        using var writer = new CodedOutputStream(ms);
        writer.WriteUInt64(1, PlayerId);
        writer.WriteUInt64(2, Seq);
        writer.WriteInt32(3, (int)Category);
        writer.WriteUInt32(4, MsgId);
        writer.WriteBytes(5, Google.Protobuf.ByteString.CopyFrom(Payload));
        writer.WriteUInt64(6, Timestamp);
        writer.Flush();
        return ms.ToArray();
    }
}

// ★ 服务端回你的消息
public class ServerEnvelope {
    public ulong Seq;
    public MsgCategory Category;
    public uint MsgId;
    public byte[] Payload;
    public uint ErrorCode;
    public string ErrorMsg;

    public static ServerEnvelope Deserialize(byte[] data) {
        var input = new CodedInputStream(data);
        var env = new ServerEnvelope();
        while (input.MaybeConsumeTag()) {
            var tag = input.LastTag;
            switch (WireFormat.GetTagFieldNumber(tag)) {
                case 1: env.Seq = input.ReadUInt64(); break;
                case 2: env.Category = (MsgCategory)input.ReadInt32(); break;
                case 3: env.MsgId = input.ReadUInt32(); break;
                case 4: env.Payload = input.ReadBytes().ToByteArray(); break;
                case 5: env.ErrorCode = input.ReadUInt32(); break;
                case 6: env.ErrorMsg = input.ReadString(); break;
            }
        }
        return env;
    }
}

public class RedirectNotify {
    public string TargetAddr;
    public uint TargetPort;
    public byte[] Token;
    public ulong BattleId;
}

// Battle.cs
public class PlayerInput {
    public ulong PlayerId, Frame;
    public List<uint> Keys;
    public Vec3 MoveDir, LookDir;
    public ulong Timestamp;

    public byte[] Serialize() {
        using var ms = new MemoryStream();
        using var w = new CodedOutputStream(ms);
        w.WriteUInt64(1, PlayerId);
        w.WriteUInt64(2, Frame);
        foreach (var k in Keys) w.WriteUInt32(3, k);
        w.WriteFloat(4, MoveDir.X); w.WriteFloat(5, MoveDir.Y); w.WriteFloat(6, MoveDir.Z);
        w.WriteFloat(7, LookDir.X); w.WriteFloat(8, LookDir.Y); w.WriteFloat(9, LookDir.Z);
        w.WriteUInt64(10, Timestamp);
        w.Flush();
        return ms.ToArray();
    }
}

public class FrameData {
    public ulong Frame;
    public List<PlayerInput> Inputs;
    public ulong ServerTime;

    public static FrameData Deserialize(byte[] data) {
        var input = new CodedInputStream(data);
        var fd = new FrameData { Inputs = new List<PlayerInput>() };
        while (input.MaybeConsumeTag()) {
            switch (WireFormat.GetTagFieldNumber(input.LastTag)) {
                case 1: fd.Frame = input.ReadUInt64(); break;
                case 2: {
                    var sub = input.ReadBytes();
                    fd.Inputs.Add(PlayerInput.Deserialize(sub.ToByteArray()));
                    break;
                }
                case 3: fd.ServerTime = input.ReadUInt64(); break;
            }
        }
        return fd;
    }
}

4. 第二步：TCP 连接与消息收发

4.1 协议格式

Ares 的 TCP 协议超简单：

┌──────────────────┬─────────────────────┐
│  4 bytes (大端)   │  Protobuf 二进制     │
│  消息体长度       │  ClientEnvelope      │
└──────────────────┴─────────────────────┘

4.2 NetworkManager——你的网络核心

using System;
using System.Collections.Generic;
using System.Net.Sockets;
using System.Threading;
using System.Threading.Tasks;
using UnityEngine;

/// <summary>
/// Ares 网络管理器 —— 挂到场景里一个 GameObject 上就能用
/// </summary>
public class AresNetworkManager : MonoBehaviour
{
    // ===== 配置 =====
    [Header("服务器地址")]
    public string GatewayHost = "127.0.0.1";
    public int GatewayPort = 9000;

    // ===== 连接状态 =====
    private TcpClient _tcpClient;
    private NetworkStream _stream;
    private Thread _receiveThread;
    private bool _isConnected;

    // ===== 消息队列 =====
    private readonly object _sendLock = new();
    private readonly Queue<byte[]> _sendQueue = new();
    private readonly Queue<ServerEnvelope> _receiveQueue = new();

    // ===== 玩家信息 =====
    public ulong PlayerId { get; set; }
    public byte[] SessionToken { get; set; }

    // ===== 事件 =====
    public event Action<ServerEnvelope> OnMessage;
    public event Action OnConnected;
    public event Action<string> OnDisconnected;
    public event Action<RedirectNotify> OnRedirectToBattle;

    void Start() => Connect();
    void OnDestroy() => Disconnect();

    /// <summary>
    /// 连接网关
    /// </summary>
    public async void Connect()
    {
        try
        {
            _tcpClient = new TcpClient();
            await _tcpClient.ConnectAsync(GatewayHost, GatewayPort);
            _stream = _tcpClient.GetStream();
            _isConnected = true;

            // 启动发送线程
            Task.Run(SendLoop);
            // 启动接收线程
            _receiveThread = new Thread(ReceiveLoop);
            _receiveThread.Start();

            Debug.Log($"[Ares] 已连接 Gateway: {GatewayHost}:{GatewayPort}");
        }
        catch (Exception e)
        {
            Debug.LogError($"[Ares] 连接失败: {e.Message}");
        }
    }

    /// <summary>
    /// ★ 发送消息 —— 你只需要调这一个方法
    /// </summary>
    public void Send(ulong playerId, MsgCategory category, uint msgId, byte[] payload)
    {
        if (!_isConnected) return;

        var envelope = new ClientEnvelope
        {
            PlayerId = playerId,
            Seq = (ulong)DateTimeOffset.UtcNow.ToUnixTimeMilliseconds(),
            Category = category,
            MsgId = msgId,
            Payload = payload,
            Timestamp = (ulong)DateTimeOffset.UtcNow.ToUnixTimeMilliseconds()
        };

        byte[] data = envelope.Serialize();
        byte[] framed = FrameMessage(data);

        lock (_sendLock)
        {
            _sendQueue.Enqueue(framed);
        }
    }

    /// <summary>
    /// ★ 取消息 —— 在 Update() 里每帧调
    /// </summary>
    public void PollMessages()
    {
        lock (_receiveQueue)
        {
            while (_receiveQueue.Count > 0)
            {
                var msg = _receiveQueue.Dequeue();
                OnMessage?.Invoke(msg);
            }
        }
    }

    // ========== 内部实现 ==========

    private byte[] FrameMessage(byte[] data)
    {
        int len = data.Length;
        byte[] framed = new byte[4 + len];
        // 大端 4 字节长度前缀
        framed[0] = (byte)(len >> 24);
        framed[1] = (byte)(len >> 16);
        framed[2] = (byte)(len >> 8);
        framed[3] = (byte)(len);
        Array.Copy(data, 0, framed, 4, len);
        return framed;
    }

    private async Task SendLoop()
    {
        while (_isConnected)
        {
            byte[] data = null;
            lock (_sendLock)
            {
                if (_sendQueue.Count > 0)
                    data = _sendQueue.Dequeue();
            }

            if (data != null && _stream != null)
            {
                try
                {
                    await _stream.WriteAsync(data, 0, data.Length);
                }
                catch { break; }
            }
            else
            {
                await Task.Delay(1);
            }
        }
    }

    private void ReceiveLoop()
    {
        byte[] header = new byte[4];
        while (_isConnected)
        {
            try
            {
                // 读 4 字节长度
                if (_stream.Read(header, 0, 4) != 4) break;
                int len = (header[0] << 24) | (header[1] << 16) | (header[2] << 8) | header[3];

                if (len <= 0 || len > 1024 * 1024) continue; // 最大 1MB

                // 读消息体
                byte[] body = new byte[len];
                int read = 0;
                while (read < len)
                {
                    int n = _stream.Read(body, read, len - read);
                    if (n == 0) throw new Exception("连接断开");
                    read += n;
                }

                // 解码
                var envelope = ServerEnvelope.Deserialize(body);
                lock (_receiveQueue)
                {
                    _receiveQueue.Enqueue(envelope);
                }
            }
            catch { break; }
        }

        _isConnected = false;
        UnityMainThreadDispatcher.Enqueue(() => OnDisconnected?.Invoke("连接断开"));
    }

    public void Disconnect()
    {
        _isConnected = false;
        _stream?.Close();
        _tcpClient?.Close();
        _receiveThread?.Join(1000);
    }
}

/// <summary>
/// Unity 主线程调度器 —— 回调事件必须在主线程执行
/// </summary>
public class UnityMainThreadDispatcher : MonoBehaviour
{
    private static readonly Queue<Action> _actions = new();
    void Update() { lock (_actions) { while (_actions.Count > 0) _actions.Dequeue()(); } }
    public static void Enqueue(Action a) { lock (_actions) _actions.Enqueue(a); }
}

5. 第三步：登录认证

5.1 登录请求

/// <summary>
/// 客户端入口 —— 挂到场景里
/// </summary>
public class GameClient : MonoBehaviour
{
    public AresNetworkManager Network;

    void Start()
    {
        Network.OnMessage += HandleMessage;
        Network.OnConnected += () => Login("test_user", "123456");
        Network.OnRedirectToBattle += OnRedirectToBattle;
    }

    /// <summary>
    /// 第一步：登录
    /// </summary>
    void Login(string account, string password)
    {
        Debug.Log($"[Client] 登录: {account}");

        // 构造 LoginRequest（按 game.proto 定义）
        byte[] payload = BuildLoginRequest(account, password, 
            SystemInfo.deviceUniqueIdentifier,
            $"{Network.GatewayHost}:{Network.GatewayPort}");

        // 发送！（注意 Category=Auth）
        Network.Send(0, MsgCategory.Auth, msgId: 1, payload);
    }

    byte[] BuildLoginRequest(string account, string password, string deviceId, string gatewayAddr)
    {
        using var ms = new MemoryStream();
        using var w = new CodedOutputStream(ms);
        w.WriteString(1, account);
        w.WriteString(2, password);
        w.WriteString(3, deviceId);
        w.WriteString(4, gatewayAddr);
        w.Flush();
        return ms.ToArray();
    }

    /// <summary>
    /// 处理所有服务端消息
    /// </summary>
    void HandleMessage(ServerEnvelope msg)
    {
        Debug.Log($"[Client] 收到消息: category={msg.Category}, msgId={msg.MsgId}");

        if (msg.ErrorCode != 0)
        {
            Debug.LogError($"[Client] 错误: {msg.ErrorMsg}");
            return;
        }

        switch (msg.Category)
        {
            case MsgCategory.Auth:
                HandleAuthResponse(msg);
                break;
            case MsgCategory.Lobby:
                HandleLobbyResponse(msg);
                break;
            case MsgCategory.Room:
                HandleRoomResponse(msg);
                break;
        }
    }

    void HandleAuthResponse(ServerEnvelope msg)
    {
        // 解析 LoginResponse
        var input = new CodedInputStream(msg.Payload);
        bool success = false;
        ulong playerId = 0;
        byte[] token = null;
        while (input.MaybeConsumeTag())
        {
            switch (WireFormat.GetTagFieldNumber(input.LastTag))
            {
                case 1: success = input.ReadBool(); break;
                case 2: playerId = input.ReadUInt64(); break;
                case 3: token = input.ReadBytes().ToByteArray(); break;
            }
        }

        if (success)
        {
            Network.PlayerId = playerId;
            Network.SessionToken = token;
            Debug.Log($"[Client] 登录成功! PlayerId={playerId}");

            // 登录成功 → 开始匹配
            StartMatch();
        }
    }
}

6. 第四步：匹配系统

6.1 发起匹配

public class GameClient : MonoBehaviour
{
    // ... 续上面的代码

    /// <summary>
    /// 第二步：开始匹配
    /// </summary>
    void StartMatch()
    {
        Debug.Log("[Client] 开始匹配...");

        // 简化版 MatchStartRequest
        byte[] payload = BuildMatchRequest(Network.PlayerId, modeId: 1, mmr: 1500);
        Network.Send(Network.PlayerId, MsgCategory.Lobby, msgId: 10, payload);
    }

    byte[] BuildMatchRequest(ulong playerId, uint modeId, uint mmr)
    {
        using var ms = new MemoryStream();
        using var w = new CodedOutputStream(ms);
        w.WriteUInt64(1, playerId);
        w.WriteUInt32(2, modeId);
        w.WriteUInt32(3, mmr);
        w.Flush();
        return ms.ToArray();
    }

    /// <summary>
    /// 定期查询匹配状态（每 1 秒）
    /// </summary>
    void PollMatchStatus()
    {
        if (_matched) return;
        byte[] payload = BuildMatchStatusRequest(Network.PlayerId);
        Network.Send(Network.PlayerId, MsgCategory.Lobby, msgId: 12, payload);
    }

    byte[] BuildMatchStatusRequest(ulong playerId)
    {
        using var ms = new MemoryStream();
        using var w = new CodedOutputStream(ms);
        w.WriteUInt64(1, playerId);
        w.Flush();
        return ms.ToArray();
    }

    void HandleLobbyResponse(ServerEnvelope msg)
    {
        // msgId=11 → MatchStartResponse
        // msgId=13 → MatchStatusResponse（包含 Redirect 信息）
        if (msg.MsgId == 13) // MatchStatusResponse
        {
            var input = new CodedInputStream(msg.Payload);
            int status = 0;
            while (input.MaybeConsumeTag())
            {
                if (WireFormat.GetTagFieldNumber(input.LastTag) == 1)
                    status = input.ReadInt32();
            }

            if (status == 2) // MATCHED
            {
                Debug.Log("[Client] 匹配成功！等待重定向...");
                _matched = true;
            }
        }
    }

    /// <summary>
    /// 第三步：收到重定向 → 直连 Battle Server
    /// </summary>
    void OnRedirectToBattle(RedirectNotify redirect)
    {
        Debug.Log($"[Client] 收到重定向 → Battle {redirect.TargetAddr}:{redirect.TargetPort}");
        // ★ 断掉 Gateway？不用！Gateway 连接留着（聊天/通知用）
        // 新开一个连接直连 Battle
        StartBattleConnection(redirect);
    }

    private bool _matched;
    private float _matchPollTimer;

    void Update()
    {
        // 匹配中定期轮询
        if (_isMatching && !_matched)
        {
            _matchPollTimer += Time.deltaTime;
            if (_matchPollTimer >= 1f)
            {
                _matchPollTimer = 0;
                PollMatchStatus();
            }
        }
    }

    bool _isMatching;
    void StartMatchPolling()
    {
        _isMatching = true;
        _matchPollTimer = 0;
    }
}

7. 第五步：进入战斗（帧同步）

7.1 帧同步核心逻辑

帧同步（Lockstep）是 Ares 的默认模式，你只需要理解一个概念：

服务器收集所有人的输入 → 打包成 FrameData → 广播给所有人

/// <summary>
/// 战斗客户端 —— 处理帧同步
/// </summary>
public class BattleClient : MonoBehaviour
{
    // ===== 配置 =====
    public int LogicFPS = 20;              // 逻辑帧率（和服务端一致）
    public int InputBufferSize = 3;        // 输入缓冲帧数

    // ===== 网络 =====
    private TcpClient _battleTcp;
    private NetworkStream _battleStream;
    private Thread _receiveThread;
    private bool _inBattle;

    // ===== 帧同步核心 =====
    private readonly Queue<FrameData> _frameQueue = new();  // 待处理的帧
    private readonly Queue<PlayerInput> _pendingInputs = new(); // 待发的输入
    private ulong _currentFrame;
    private float _logicTimer;

    // ===== 你的游戏逻辑引用 =====
    public IGameLogic GameLogic; // ← 你的游戏逻辑接口

    /// <summary>
    /// 连接 Battle Server
    /// </summary>
    public void Connect(string host, int port, byte[] joinToken)
    {
        _battleTcp = new TcpClient();
        _battleTcp.Connect(host, port);
        _battleStream = _battleTcp.GetStream();
        _inBattle = true;

        // 发送 BattleJoinRequest（含 token）
        SendJoinRequest(joinToken);

        // 启动接收
        _receiveThread = new Thread(BattleReceiveLoop);
        _receiveThread.Start();

        Debug.Log($"[Battle] 已连接 Battle Server: {host}:{port}");
    }

    /// <summary>
    /// ★ 发送玩家输入（每帧调）
    /// </summary>
    public void SendInput(List<uint> keys, Vec3 moveDir, Vec3 lookDir)
    {
        var input = new PlayerInput
        {
            PlayerId = 0, // Battle Server 知道你是谁
            Frame = _currentFrame + (ulong)InputBufferSize, // 预发送 N 帧后的输入
            Keys = keys,
            MoveDir = moveDir,
            LookDir = lookDir,
            Timestamp = (ulong)DateTimeOffset.UtcNow.ToUnixTimeMilliseconds()
        };

        _pendingInputs.Enqueue(input);
    }

    /// <summary>
    /// ★ 主循环 —— 在 Unity Update() 里每帧调
    /// </summary>
    void Update()
    {
        if (!_inBattle) return;

        // ===== 1. 收集并发送输入 =====
        var keys = new List<uint>();
        if (Input.GetKey(KeyCode.W)) keys.Add(1);
        if (Input.GetKey(KeyCode.S)) keys.Add(2);
        if (Input.GetKey(KeyCode.A)) keys.Add(3);
        if (Input.GetKey(KeyCode.D)) keys.Add(4);
        if (Input.GetKeyDown(KeyCode.Space)) keys.Add(5); // 跳跃

        var moveDir = new Vec3 {
            X = Input.GetAxis("Horizontal"),
            Y = 0,
            Z = Input.GetAxis("Vertical")
        };

        SendInput(keys, moveDir, new Vec3());

        // 发送队列中的输入
        FlushInputs();

        // ===== 2. 推进逻辑帧 =====
        _logicTimer += Time.deltaTime;
        float frameInterval = 1f / LogicFPS;

        while (_logicTimer >= frameInterval)
        {
            _logicTimer -= frameInterval;
            TickLogic();
        }

        // ===== 3. 渲染插值 =====
        float t = _logicTimer / frameInterval;
        GameLogic?.Render(t);
    }

    /// <summary>
    /// 推进一帧游戏逻辑
    /// </summary>
    void TickLogic()
    {
        FrameData frameData = null;
        lock (_frameQueue)
        {
            if (_frameQueue.Count > 0)
                frameData = _frameQueue.Dequeue();
        }

        if (frameData == null) return; // 还没收到帧数据

        _currentFrame = frameData.Frame;

        // ★ 把所有玩家的输入交给游戏逻辑
        foreach (var input in frameData.Inputs)
        {
            GameLogic?.ApplyInput(input.PlayerId, input.Keys, input.MoveDir);
        }

        // ★ 执行确定性演算
        GameLogic?.Tick();
    }

    // ========== 内部实现 ==========

    void FlushInputs()
    {
        if (_battleStream == null) return;
        lock (_pendingInputs)
        {
            while (_pendingInputs.Count > 0)
            {
                var input = _pendingInputs.Dequeue();
                byte[] data = input.Serialize();
                byte[] framed = FrameMessage(data);
                _battleStream.Write(framed, 0, framed.Length);
            }
        }
    }

    void SendJoinRequest(byte[] token)
    {
        using var ms = new MemoryStream();
        using var w = new CodedOutputStream(ms);
        w.WriteUInt64(1, 0); // player_id
        w.WriteBytes(2, Google.Protobuf.ByteString.CopyFrom(token));
        w.Flush();
        byte[] data = ms.ToArray();
        byte[] framed = FrameMessage(data);
        _battleStream.Write(framed, 0, framed.Length);
    }

    void BattleReceiveLoop()
    {
        byte[] header = new byte[4];
        while (_inBattle)
        {
            try
            {
                if (_battleStream.Read(header, 0, 4) != 4) break;
                int len = (header[0] << 24) | (header[1] << 16) | (header[2] << 8) | header[3];
                byte[] body = new byte[len];
                int read = 0;
                while (read < len) { read += _battleStream.Read(body, read, len - read); }

                var fd = FrameData.Deserialize(body);
                lock (_frameQueue) { _frameQueue.Enqueue(fd); }
            }
            catch { break; }
        }
    }

    byte[] FrameMessage(byte[] data)
    {
        byte[] framed = new byte[4 + data.Length];
        framed[0] = (byte)(data.Length >> 24);
        framed[1] = (byte)(data.Length >> 16);
        framed[2] = (byte)(data.Length >> 8);
        framed[3] = (byte)(data.Length);
        Array.Copy(data, 0, framed, 4, data.Length);
        return framed;
    }

    void OnDestroy()
    {
        _inBattle = false;
        _battleStream?.Close();
        _battleTcp?.Close();
    }
}

/// <summary>

public class SimpleMoveLogic : MonoBehaviour, IGameLogic

    private Dictionary<ulong, Vector3> _positions = new();

    public GameObject PlayerPrefab; // 在 Inspector 里拖入

    public void ApplyInput(ulong playerId, List<uint> keys, Vec3 moveDir)
    {
        if (!_positions.ContainsKey(playerId))
            _positions[playerId] = Vector3.zero;

        float speed = 5f;
        Vector3 move = new Vector3(moveDir.X, moveDir.Y, moveDir.Z) * speed * (1f / 20f);
        _positions[playerId] += move;
    }

    public void Tick()
    {
        // 保存上一帧位置（用于插值）
        foreach (var kv in _positions)
            _prevPositions[kv.Key] = kv.Value;
    }

    {
        // 插值显示
        foreach (var kv in _positions)
        {
            Vector3 prev = _prevPositions.ContainsKey(kv.Key) ? _prevPositions[kv.Key] : kv.Value;
            Vector3 display = Vector3.Lerp(prev, kv.Value, t);
            // 更新 GameObject 位置...
        }
    }
}

Q2: 登录返回失败

可能原因：
- account 为空 → 必须填字符串
- Gateway 还没连上 Game Server → 先启动 Game 再启动 Gateway

Q3: 匹配后收不到 Redirect

Ares 当前的匹配流程：
1. 发 StartMatch → 加入匹配池
2. 定期发 GetMatchStatus（每 1 秒）
3. 状态变成 MATCHED 后 → 会通过 ServerEnvelope 收到重定向信息
4. 用 Redirect 信息直连 Battle

float t = _logicTimer / frameInterval; // 0~1 之间
